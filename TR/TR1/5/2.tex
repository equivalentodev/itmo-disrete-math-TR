\\
\\

\begin{equation*}
    \overline{\overline{A} \cap \overline{B}} \cup (A \cup \overline{\overline{B} \cap \overline{C}} \cup \overline{\overline{A} \cap \overline{B} \cap \overline{C}}) \cup P\cup ((A\cap B\cap \overline{C})\cup (A \cap (\overline{B}\cup C)))
\end{equation*}
